
#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int subarraySum(vector<int>& nums, int k) {
    unordered_map<int, int> mp;

    int prefixSum = 0;
    int count = 0;

    mp[0] = 1;

    for (int num : nums) {
        prefixSum += num;

        int remaining = prefixSum - k;

        if (mp.find(remaining) != mp.end()) {
            count += mp[remaining];
        }

        mp[prefixSum]++;
    }

    return count;
}

int main() {
    int n, k;

    cout << "Enter array size: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter array elements: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    cout << "Enter k: ";
    cin >> k;

    cout << "Number of subarrays: " << subarraySum(nums, k);

    return 0;
}
