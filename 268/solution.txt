
#include <iostream>
#include <vector>
using namespace std;

int missingNumber(vector<int>& nums) {
    int n = nums.size();

    int sum = n * (n + 1) / 2;
    int arrSum = 0;

    for (int i = 0; i < n; i++) {
        arrSum += nums[i];
    }

    return sum - arrSum;
}

int main() {
    vector<int> nums = {3, 0, 1};

    cout << "Missing Number: " << missingNumber(nums);

    return 0;
}
